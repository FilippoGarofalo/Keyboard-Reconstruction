 1. STRUTTURA AD ALTO LIVELLO
===========================================================================

  ┌──────────────────────────────────────────────────────────────────┐
  │  CALIBRAZIONE (una volta sola, prima del loop)                   │
  │    load H_white_calibration.mat → calibration, virtualKeyboard   │
  └──────────────────────────────────────────────────────────────────┘
                              │
                              ▼
  ┌──────────────────────────────────────────────────────────────────┐
  │  LOOP SUI FRAME DEL VIDEO                                        │
  │                                                                  │
  │   frame ──► [Filippo]  getFingers(frame)                         │
  │                │                                                 │
  │                ▼                                                 │
  │           fingertips (Nx2 pixel originali)                       │
  │                │                                                 │
  │                ▼                                                 │
  │           [Flavio]  map_image_point_to_key(...)                  │
  │                │                                                 │
  │                ▼                                                 │
  │           keyIndices (Nx1, 1..K oppure 0)                        │
  │                                                                  │
  │   prev,curr ──► [Emanuele] detect_key_activity(...)              │
  │                │                                                 │
  │                ▼                                                 │
  │           keyActivity (1xK)                                      │
  │                                                                  │
  │   FUSIONE: tasto i premuto se                                    │
  │     (qualche dito mappa su i)  AND  (keyActivity(i) > soglia)    │
  └──────────────────────────────────────────────────────────────────┘


===========================================================================
 2. ESEMPIO DI SCRIPT (pseudo-MATLAB)
===========================================================================

  % --- Setup paths --------------------------------------------------------
  addpath('src/convex_hull');                       % Filippo
  addpath('src/keyboard model/homography_white');   % Flavio (calibrazione)
  addpath('src/keyboard model/virtual_keyboard');   % Flavio (modello tasti)
  addpath('src/keyboard model/mapping');            % Flavio (mapping)
  addpath('src/transform');                         % Emanuele

  % --- Calibrazione (una volta) -------------------------------------------
  load('H_white_calibration.mat', 'calibration');
  vk = calibration.virtualKeyboard;
  threshold = 5.0;   % soglia su keyActivity, da tarare

  % --- Apertura video -----------------------------------------------------
  video = VideoReader('video.mp4');
  prevFrame = [];

  % --- Loop principale ----------------------------------------------------
  while hasFrame(video)
      frame = readFrame(video);

      % [Filippo] Rileva le punte delle dita
      fingertips = getFingers(frame);                           % Nx2 px

      % [Flavio] Mappa i polpastrelli sui tasti bianchi
      keyIndices = map_image_point_to_key( ...
          fingertips, calibration.H_image_to_plane, vk);        % Nx1 uint32

      % [Emanuele] Stima quanto è cambiato ciascun tasto
      if ~isempty(prevFrame)
          keyActivity = detect_key_activity( ...
              prevFrame, frame, calibration.roiMask, vk, calibration);
      else
          keyActivity = zeros(1, vk.numKeys);
      end

      % --- Fusione: dito + attività -> tasto premuto ----------------------
      pressedKeys = false(1, vk.numKeys);
      for i = 1:numel(keyIndices)
          ki = keyIndices(i);
          if ki > 0 && keyActivity(ki) > threshold
              pressedKeys(ki) = true;
          end
      end

      % --- Output ---------------------------------------------------------
      pressedIdx = find(pressedKeys);
      for j = 1:numel(pressedIdx)
          fprintf('Tasto premuto: %s\n', vk.keys(pressedIdx(j)).name);
      end

      prevFrame = frame;
  end
